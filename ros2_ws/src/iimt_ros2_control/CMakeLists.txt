cmake_minimum_required(VERSION 3.16)
project(iimt_ros2_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(action_msgs REQUIRED)
find_package(Threads REQUIRED)
find_package(iimt_ros2_driver REQUIRED)
add_library(iimt_system SHARED src/system.cpp src/transport.cpp)
target_include_directories(iimt_system PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(iimt_system hardware_interface pluginlib rclcpp rclcpp_lifecycle diagnostic_msgs rcl_interfaces iimt_ros2_driver action_msgs)
target_link_libraries(iimt_system ${CMAKE_DL_LIBS} Threads::Threads)
target_compile_options(iimt_system PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(hardware_interface iimt_system.xml)
install(TARGETS iimt_system EXPORT export_iimt_system LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_subdirectory(test)
endif()
ament_export_targets(export_iimt_system HAS_LIBRARY_TARGET)
ament_export_dependencies(hardware_interface pluginlib rclcpp rclcpp_lifecycle diagnostic_msgs rcl_interfaces iimt_ros2_driver action_msgs)
ament_package()
